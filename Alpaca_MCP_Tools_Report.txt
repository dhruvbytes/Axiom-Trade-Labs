3. place_option_order
Description: Place an options order (single-leg or multi-leg).

For single-leg orders, provide symbol, side, and qty.
For multi-leg orders, provide qty, legs, and optionally
order_class="mleg" (auto-inferred). Symbol and side on the
parent are not needed for multi-leg.

{
  "additionalProperties": false,
  "properties": {
    "qty": {
      "type": "string",
      "description": "Number of contracts. Required for both single-leg and\n multi-leg orders. For multi-leg, this is the strategy\n multiplier \u2014 each leg's ratio_qty is scaled by this\n value (e.g., qty=\"10\" with ratio_qty=\"2\" = 20\n contracts for that leg)."
    },
    "type": {
      "default": "market",
      "type": "string",
      "description": "\"market\" or \"limit\"."
    },
    "time_in_force": {
      "default": "day",
      "type": "string",
      "description": "\"day\" only. Options do not support other\n           values."
    },
    "symbol": {
      "anyOf": [
        {
          "type": "string"
        },
        {
          "type": "null"
        }
      ],
      "default": null,
      "description": "OCC option symbol (e.g., \"AAPL250321C00150000\").\n    Required for single-leg."

    "side": {

        {
          "type": "null"

    "position_intent": {

      ],
      "default": null,
